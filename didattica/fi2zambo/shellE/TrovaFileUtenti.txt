# ese11b

for i in * #cioe per ogni file del direttorio
do
if [ -d $i ]
then
  for j in $* #cioe per ogni argomento passato
  do
    if [ $i = $j ]
    then cd $i;ese11c $*;cd ..
    else cd $i; ese11b $*;cd ..
    fi
  done
fi
done
