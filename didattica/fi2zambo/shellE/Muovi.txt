 o|1) echo due argomenti; exit;;

shift

echo $diret $1

  for i in $*
  do

      read risposta
      if test $risposta = "si" -o $risposta = "s"
	then
	  if mv $i $diret
	    # il risultato della move comanda la alternativa
	    then echo si $1/$diret
	    else echo problemi per $diret
          fi
      fi
  done
